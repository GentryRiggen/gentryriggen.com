/* {{TITLE}} service worker: lets the home-screen app work offline.
 *
 * Generated from lib/game-kit/serviceWorker/template.sw.txt by
 * `npx tsx scripts/generate-game-sw.ts` (edit the template, not this file).
 * Registered by components/game-kit/useGameServiceWorker.ts with scope
 * "/{{SLUG}}", so it only controls the {{TITLE}} page. Rules:
 * - Only same-origin GET requests are ever handled; everything else goes
 *   straight to the network untouched.
 * - Navigations to the {{TITLE}} page: network-first, falling back to the
 *   cached page when offline or slow (3s). When a deploy changes the files the
 *   page references, cached /_next/static/ files it no longer uses are pruned.
 * - /_next/static/* (content-hashed, immutable): cache-first.
 * - Game assets that are not content-hashed (ASSET_PREFIXES and ASSET_PATHS
 *   below): network-first, falling back to the cached copy when offline or slow (3s);
 *   stored without the query.
 * - Only `ok`, same-origin (`type === "basic"`), non-redirected responses are
 *   cached, so opaque or cross-origin responses never enter the cache.
 * - /{{SLUG}}/version.json is never handled, cached or precached.
 * - The page posts { type: "SKIP_WAITING" } to make a waiting worker take over.
 * - The page posts { type: "CACHE_URLS", urls } with the resources it has
 *   already loaded, so a single online visit is enough to play offline.
 */

const CACHE_PREFIX = "{{SLUG}}-";
const CACHE = `${CACHE_PREFIX}v1`;
const PAGE_PATH = "/{{SLUG}}";
const STATIC_PREFIX = "/_next/static/";
/** Folders of non-hashed game assets, handled network-first. */
const ASSET_PREFIXES = {{ASSET_PREFIXES}};
/** Single non-hashed files (for example the manifest), handled the same way. */
const ASSET_PATHS = {{ASSET_PATHS}};
/** Upper bound on URLs accepted from one CACHE_URLS message. */
const MAX_PRECACHE_URLS = 300;

// No skipWaiting on install: a new worker waits while a game is being played,
// and takes over only when the page asks (SKIP_WAITING, after a tap on the
// update pill). A first install has nothing to wait for and activates alone.

self.addEventListener("activate", (event) => {
  event.waitUntil(
    (async () => {
      const keys = await caches.keys();
      await Promise.all(
        keys
          .filter((key) => key.startsWith(CACHE_PREFIX) && key !== CACHE)
          .map((key) => caches.delete(key))
      );
      await self.clients.claim();
    })()
  );
});

/** True for the game page itself ("/{{SLUG}}", "/{{SLUG}}.html"). */
function isGamePath(pathname) {
  return (
    pathname === PAGE_PATH ||
    pathname === `${PAGE_PATH}.html` ||
    pathname === `${PAGE_PATH}/`
  );
}

/** True for a non-hashed game asset (see ASSET_PREFIXES and ASSET_PATHS). */
function isGameAsset(pathname) {
  return (
    ASSET_PATHS.includes(pathname) ||
    ASSET_PREFIXES.some((prefix) => pathname.startsWith(prefix))
  );
}

/** True for the game page or one of its sub-pages (e.g. /{{SLUG}}/versions). */
function isPagePath(pathname) {
  if (isGameAsset(pathname) || isVersionFile(pathname)) return false;
  return (
    pathname === PAGE_PATH ||
    pathname === `${PAGE_PATH}.html` ||
    pathname.startsWith(`${PAGE_PATH}/`)
  );
}

/**
 * The deployed-version file the page polls for updates. It must always come
 * from the network, so the worker never stores or serves it.
 */
function isVersionFile(pathname) {
  return pathname === `${PAGE_PATH}/version.json`;
}

function isStaticAsset(pathname) {
  return pathname.startsWith(STATIC_PREFIX);
}

function isCacheable(response) {
  return (
    Boolean(response) &&
    response.ok &&
    response.type === "basic" &&
    !response.redirected
  );
}

/** Parses a same-origin URL, or returns null for anything else. */
function toSameOriginUrl(value) {
  if (typeof value !== "string") return null;
  try {
    const url = new URL(value, self.location.origin);
    return url.origin === self.location.origin ? url : null;
  } catch {
    return null;
  }
}

/** Pages are stored without query/hash, plus under the canonical page path. */
async function cachePage(cache, url, response) {
  const key = `${url.origin}${url.pathname}`;
  const puts = [cache.put(key, response.clone())];
  // Only the game itself doubles as the offline fallback; a sub-page such as
  // the version history must never stand in for it.
  if (isGamePath(url.pathname)) {
    puts.push(cache.put(new URL(PAGE_PATH, url.origin).href, response.clone()));
  }
  await Promise.all(puts);
}

/** How long a page waits on the network before the cached page is used. */
const NAVIGATION_NETWORK_TIMEOUT_MS = 3000;

/** Matches every /_next/static/ path in a page's HTML. */
const STATIC_REF_PATTERN = /\/_next\/static\/[^"'\s\\)<>?]+/g;

function staticRefs(html) {
  return new Set(html.match(STATIC_REF_PATTERN) ?? []);
}

function sameSet(a, b) {
  if (a.size !== b.size) return false;
  for (const value of a) if (!b.has(value)) return false;
  return true;
}

/**
 * A new deploy changes the hashed files the game page references, which
 * would otherwise stay in the cache forever. When the page's references
 * change, drop only the cached /_next/static/ files the previous page
 * referenced and the new page dropped. Lazily loaded chunks are never named in
 * the HTML, so they stay: an offline feature opened once must keep working
 * after a deploy. Does nothing when the references cannot be read or have not
 * changed.
 */
async function pruneStaleStatic(cache, previous, response) {
  let before;
  let after;
  try {
    before = staticRefs(await previous.clone().text());
    after = staticRefs(await response.clone().text());
  } catch {
    return;
  }
  if (after.size === 0 || sameSet(before, after)) return;
  for (const request of await cache.keys()) {
    const { pathname } = new URL(request.url);
    if (before.has(pathname) && !after.has(pathname)) {
      await cache.delete(request);
    }
  }
}

/**
 * Network-first, but a slow network (lie-fi) falls back to the cached page
 * after a few seconds. A late answer still updates the cache.
 */
async function handleNavigation(event, request, url) {
  const cache = await caches.open(CACHE);
  const key = `${url.origin}${url.pathname}`;
  const network = fetch(request).then(async (response) => {
    if (isCacheable(response)) {
      try {
        const previous = isGamePath(url.pathname)
          ? await cache.match(key)
          : undefined;
        await cachePage(cache, url, response);
        if (previous) await pruneStaleStatic(cache, previous, response);
      } catch {
        // Caching is best effort.
      }
    }
    return response;
  });
  event.waitUntil(network.catch(() => {}));
  const cached =
    (await cache.match(key)) ??
    (await cache.match(new URL(PAGE_PATH, url.origin).href));
  if (!cached) return network;
  const slow = new Promise((resolve) =>
    setTimeout(() => resolve(cached), NAVIGATION_NETWORK_TIMEOUT_MS)
  );
  // A 404 or 5xx is as good as no answer: the cached page is better.
  const usable = network.then((response) => (response.ok ? response : cached));
  return Promise.race([usable.catch(() => cached), slow]);
}

async function handleStaticAsset(request) {
  const cache = await caches.open(CACHE);
  const cached = await cache.match(request);
  if (cached) return cached;
  const response = await fetch(request);
  if (isCacheable(response)) {
    await cache.put(request, response.clone()).catch(() => {});
  }
  return response;
}

/** How long an asset waits on the network before the cached copy is used. */
const ASSET_NETWORK_TIMEOUT_MS = 3000;

/**
 * Assets can change between releases: fresh when the network answers in time,
 * the cached copy when it is offline or slow. A slow response still updates
 * the cache when it arrives.
 */
async function handleGameAsset(event, request, url) {
  const cache = await caches.open(CACHE);
  const key = `${url.origin}${url.pathname}`;
  let stored = Promise.resolve();
  const network = fetch(request).then((response) => {
    if (isCacheable(response)) {
      // The caller gets the response now; the cache write finishes under
      // waitUntil below.
      stored = cache.put(key, response.clone()).catch(() => {});
    }
    return response;
  });
  // Keep the worker alive until the network answers and the cache is updated,
  // even after a slow one has been answered from the cache.
  event.waitUntil(network.then(() => stored).catch(() => {}));
  const cached = await cache.match(key);
  if (!cached) return network;
  const slow = new Promise((resolve) =>
    setTimeout(() => resolve(cached), ASSET_NETWORK_TIMEOUT_MS)
  );
  // A 404 or 5xx is as good as no answer: the cached copy is better.
  const usable = network.then((response) => (response.ok ? response : cached));
  return Promise.race([usable.catch(() => cached), slow]);
}

self.addEventListener("fetch", (event) => {
  const { request } = event;
  if (request.method !== "GET") return;
  const url = toSameOriginUrl(request.url);
  if (!url) return;
  if (isVersionFile(url.pathname)) return;

  if (request.mode === "navigate") {
    if (isPagePath(url.pathname)) {
      event.respondWith(handleNavigation(event, request, url));
    }
    return;
  }

  // Range requests (media) need partial responses; leave them to the network.
  if (request.headers.has("range")) return;
  if (isStaticAsset(url.pathname)) {
    event.respondWith(handleStaticAsset(request));
  } else if (isGameAsset(url.pathname)) {
    event.respondWith(handleGameAsset(event, request, url));
  }
  // Everything else passes through to the network.
});

async function precache(urls) {
  const cache = await caches.open(CACHE);
  await Promise.all(
    urls.map(async (url) => {
      try {
        const response = await fetch(url.href, { credentials: "same-origin" });
        if (!isCacheable(response)) return;
        if (isPagePath(url.pathname)) {
          await cachePage(cache, url, response);
        } else {
          await cache.put(url.href, response);
        }
      } catch {
        // Ignore individual failures; the page still works online.
      }
    })
  );
}

self.addEventListener("message", (event) => {
  const { data, source } = event;
  // Only accept messages from our own pages (window clients are same-origin).
  if (!source || source.type !== "window") return;
  // The page asks a waiting worker to take over when the player taps Update.
  if (data && data.type === "SKIP_WAITING") {
    self.skipWaiting();
    return;
  }
  if (!data || data.type !== "CACHE_URLS" || !Array.isArray(data.urls)) return;

  const seen = new Set();
  const urls = [];
  for (const value of data.urls.slice(0, MAX_PRECACHE_URLS)) {
    const url = toSameOriginUrl(value);
    if (!url) continue;
    const isAsset = isGameAsset(url.pathname);
    if (!isPagePath(url.pathname) && !isStaticAsset(url.pathname) && !isAsset) {
      continue;
    }
    url.hash = "";
    if (isPagePath(url.pathname) || isAsset) url.search = "";
    if (seen.has(url.href)) continue;
    seen.add(url.href);
    urls.push(url);
  }

  if (urls.length > 0) event.waitUntil(precache(urls));
});
